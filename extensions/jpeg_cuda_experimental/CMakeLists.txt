cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(jpeg_cuda_experimental LANGUAGES CXX CUDA)
    option(BUILD_TEST "Build component tests" ON)
    enable_testing()
endif()
find_package(CUDAToolkit REQUIRED)
add_library(jpeg_cuda_experimental STATIC jpeg_header.cpp decoder.cu)
target_compile_features(jpeg_cuda_experimental PUBLIC cxx_std_17)
target_include_directories(jpeg_cuda_experimental PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(jpeg_cuda_experimental PUBLIC CUDA::cudart)
target_compile_options(jpeg_cuda_experimental PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
set_target_properties(jpeg_cuda_experimental PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_STANDARD 17)

if(BUILD_TEST)
    add_executable(jpeg_cuda_header_test tests/header_test.cpp)
    target_link_libraries(jpeg_cuda_header_test PRIVATE jpeg_cuda_experimental)
    add_test(NAME jpeg_cuda_header_test COMMAND jpeg_cuda_header_test)
    add_library(jpeg_cuda_test_bridge SHARED tests/bridge.cpp)
    target_link_libraries(jpeg_cuda_test_bridge PRIVATE jpeg_cuda_experimental)
endif()
